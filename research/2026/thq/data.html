import matplotlib.pyplot as plt
import pandas as pd

statement = financials.income_statement()
income_df = statement.to_dataframe()
income_df = income_df[['label', '2012-03-31', '2011-03-31', '2010-03-31']].copy()
income_df['label'] = income_df['label'].str.strip()

# Combine software and license royalty costs into one category
royalty_cost = income_df[income_df['label'].isin([
    'Software amortization and royalties',
    'License amortization and royalties'
])].copy()
royalty_cost = royalty_cost.set_index('label')
royalty_cost = royalty_cost.loc[['Software amortization and royalties', 'License amortization and royalties']]

cogs = income_df[income_df['label'] == 'Cost of Goods Sold'].iloc[0]
royalty_total = [
    royalty_cost['2012-03-31'].sum(),
    royalty_cost['2011-03-31'].sum(),
    royalty_cost['2010-03-31'].sum(),
]
non_royalty_cogs = [
    abs(cogs['2012-03-31']) - royalty_total[0],
    abs(cogs['2011-03-31']) - royalty_total[1],
    abs(cogs['2010-03-31']) - royalty_total[2],
]

years = ['2012', '2011', '2010']
royalty_total_thousands = [value / 1000 for value in royalty_total]
non_royalty_cogs_thousands = [value / 1000 for value in non_royalty_cogs]

# Percent of each bar that is royalty cost
royalty_pct = [
    (royalty_total[i] / (royalty_total[i] + non_royalty_cogs[i])) * 100
    for i in range(len(years))
]
other_pct = [100 - p for p in royalty_pct]

plt.figure(figsize=(10, 6))
bar_container = plt.bar(
    years,
    non_royalty_cogs_thousands,
    label='Other COGS',
    color='steelblue',
    alpha=0.8
)
plt.bar(
    years,
    royalty_total_thousands,
    bottom=non_royalty_cogs_thousands,
    label='Royalty Cost',
    color='purple',
    alpha=0.8
)

# Place percentage labels inside the stacked bar segments
for i, (bar, value) in enumerate(zip(bar_container, non_royalty_cogs_thousands)):
    plt.text(
        i,
        value / 2,
        f'{other_pct[i]:.1f}%',
        ha='center',
        va='center',
        color='white',
        fontsize=10,
        weight='bold'
    )

for i, value in enumerate(royalty_total_thousands):
    plt.text(
        i,
        non_royalty_cogs_thousands[i] + value / 2,
        f'{royalty_pct[i]:.1f}%',
        ha='center',
        va='center',
        color='white',
        fontsize=10,
        weight='bold'
    )

plt.xlabel('Fiscal Year')
plt.ylabel('Cost of Goods Sold (in thousands)')
plt.title('THQ Cost of Goods Sold Breakdown with Royalty Cost')
plt.legend()
plt.grid(True, axis='y', linestyle='--', alpha=0.3)
plt.tight_layout()
plt.show()
